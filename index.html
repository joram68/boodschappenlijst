function doGet() { return HtmlService .createHtmlOutputFromFile("Index") .setTitle("Boodschappenlijst") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function haalBoodschappenOp() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; var laatsteRij = sheet.getLastRow(); if (laatsteRij === 0) { return []; } var gegevens = sheet.getRange(1, 1, laatsteRij, 2).getValues(); var items = []; gegevens.forEach(function(rij) { var winkel = String(rij[0]).trim(); var tekst = String(rij[1]).trim(); if ( winkel && tekst && ["ALDI", "LIDL", "COLRUYT", "AH"].indexOf(winkel) !== -1 ) { items.push({ winkel: winkel, tekst: tekst }); } }); return items; } function voegBoodschapToe(winkel, tekst) { var toegestaneWinkels = ["ALDI", "LIDL", "COLRUYT", "AH"]; if (toegestaneWinkels.indexOf(winkel) === -1) { throw new Error("Ongeldige winkel."); } if (!tekst) { throw new Error("Geen boodschap opgegeven."); } var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; sheet.appendRow([ winkel, tekst ]); return true; } function verwijderBoodschap(winkel, tekst) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; var laatsteRij = sheet.getLastRow(); if (laatsteRij === 0) { return false; } var gegevens = sheet.getRange(1, 1, laatsteRij, 2).getValues(); for (var i = gegevens.length - 1; i >= 0; i--) { if ( String(gegevens[i][0]).trim() === winkel && String(gegevens[i][1]).trim() === tekst ) { sheet.deleteRow(i + 1); return true; } } return false; }
